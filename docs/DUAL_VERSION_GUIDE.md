```swift
// ????
enum BuildEdition {
    case standard
    case pro
}

// ??????????????
let CURRENT_EDITION: BuildEdition = .pro

// Feature Flags
struct FeatureFlags {
    static var tabsEnabled: Bool {
        switch CURRENT_EDITION {
        case .standard: return false
        case .pro: return true
        }
    }
    
    static var maxTabs: Int {
        switch CURRENT_EDITION {
        case .standard: return 1
        case .pro: return 10
        }
    }
    
    static var appName: String {
        switch CURRENT_EDITION {
        case .standard: return "Parquet View"
        case .pro: return "Parquet View Pro"
        }
    }
    
    static var bundleIdentifier: String {
        switch CURRENT_EDITION {
        case .standard: return "com.parquetview.app"
        case .pro: return "com.parquetview.pro"
        }
    }
}
```
